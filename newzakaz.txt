// URL: POST https://<сервер-1С>/<база>/hs/Zakaz/NewZakaz
//
// Контракт запроса/ответа: docs/ТЗ_HTTP_СЕРВИС_ЗАГРУЗКА_ЗАКАЗА_1С.md.
// Логика загрузки — порт обработки асЗагрузкаЗаказаИзJSON
// (tools/as_order_loader/as_order_loader.bsl, заказы №852/№860 проверены
// end-to-end) + создание Характеристики из tools/as_order_loader/
// build_execute_payload.py (обход бага «Оформить заказ» без характеристики,
// см. docs/BACKLOG.md). Изменения вносятся в ОБА места.
//
// Отличия от первой версии модуля (19.09.2026):
//   - штатный расчёт цен (асВводНовогоЗаказа.РедактироватьВсеНаКлиенте_ДляИзмененныхВМодуле),
//     производные размеры из регистра асНастройкиПродукции;
//   - поиск продукта через Справочники.асПродукция.Артикул (было: Номенклатура.Артикул),
//     заполнение Продукт/АртикулПродукции/АртикулГруппы/ГруппаПродукцииСтр/СечениеСтр;
//   - thickness -> t/t2, shina -> МатериалНаСоединение0/1, np, Материал2;
//   - создание/поиск Характеристики номенклатуры;
//   - order_comment -> Комментарий документа;
//   - request_id -> идемпотентность (повторный вызов вернёт ранее созданный
//     документ; маркер в комментарии «[req:...]» — тот же формат, что в
//     tools/as_order_loader/extension/ДополнениеHTTP.bsl);
//   - ответ по ТЗ: {Успех, НомерЗаказа, Ошибки, Предупреждения}, UTF-8;
//   - заголовок X-API-Key (совместимость: также принимается устаревший «key»);
//   - транзакция с откатом при исключении.
//
// ПРАВА: роль сервисного пользователя (HTTP_Service1 / АСУ_HTTPСервисы)
// должна включать: Использование HTTP-сервиса «ПолучениеЗаказов», документ
// асСпецификацияЗаказа (добавление/запись), справочники асПродукция,
// асМатериалы, асСоединениеВоздуховодов, ХарактеристикиНоменклатуры,
// Номенклатура, регистр сведений асНастройкиПродукции, обработку
// асВводНовогоЗаказа. Без права на сам HTTP-сервис платформа отвечает 403
// «Недостаточно прав для использования ресурса с данным HTTP методом».

Функция NewZakazPost(Запрос)

	Ответ = Новый HTTPСервисОтвет(200);
	Ответ.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");

	Результат = Новый Структура("Успех, НомерЗаказа, Ошибки, Предупреждения",
		Ложь, "", Новый Массив, Новый Массив);

	// --- Аутентификация по ключу (ТЗ п. 7) ---
	Если Не ПроверитьКлючAPI(Запрос) Тогда
		Ответ.КодСостояния = 403;
		Результат.Ошибки.Добавить("Нет установленного ключа обработки (X-API-Key)");
		Возврат ОтветСJSON(Ответ, Результат);
	КонецЕсли;

	// --- Чтение тела ---
	СтрокаЗапроса = Запрос.ПолучитьТелоКакСтроку(КодировкаТекста.UTF8);

	Данные = Неопределено;
	Попытка
		Чтение = Новый ЧтениеJSON;
		Чтение.УстановитьСтроку(СтрокаЗапроса);
		Данные = ПрочитатьJSON(Чтение, Истина); // соответствиями — устойчиво к любым ключам
	Исключение
		Данные = Неопределено;
	КонецПопытки;

	Если Данные = Неопределено Тогда
		Ответ.КодСостояния = 500;
		Результат.Ошибки.Добавить("Не удалось разобрать JSON.");
		Возврат ОтветСJSON(Ответ, Результат);
	КонецЕсли;

	// --- Извлечение positions / order_comment / request_id (ТЗ п. 3) ---
	КомментарийЗаказа = "Загрузка из JSON";
	ИдЗапроса = "";
	Если ТипЗнч(Данные) = Тип("Соответствие") Тогда
		Позиции = Данные.Получить("positions");
		Если ЗначениеЗаполнено(Данные.Получить("order_comment")) Тогда
			КомментарийЗаказа = Данные.Получить("order_comment");
		КонецЕсли;
		ИдЗапроса = Данные.Получить("request_id");
		Если ИдЗапроса = Неопределено Тогда
			ИдЗапроса = "";
		КонецЕсли;
	ИначеЕсли ТипЗнч(Данные) = Тип("Массив") Тогда
		Позиции = Данные;
	Иначе
		Ответ.КодСостояния = 500;
		Результат.Ошибки.Добавить("Неизвестная структура JSON");
		Возврат ОтветСJSON(Ответ, Результат);
	КонецЕсли;

	Если ТипЗнч(Позиции) <> Тип("Массив") Тогда
		Ответ.КодСостояния = 500;
		Результат.Ошибки.Добавить("Не найден массив positions");
		Возврат ОтветСJSON(Ответ, Результат);
	КонецЕсли;

	Если Позиции.Количество() = 0 Тогда
		Ответ.КодСостояния = 400;
		Результат.Ошибки.Добавить("Пустой массив positions");
		Возврат ОтветСJSON(Ответ, Результат);
	КонецЕсли;

	// --- Загрузка ---
	Попытка
		ЗагрузитьЗаказ(Результат, Позиции, КомментарийЗаказа, ИдЗапроса);
	Исключение
		Ответ.КодСостояния = 500;
		Результат.Успех = Ложь;
		Результат.Ошибки.Добавить("Запись документа: " + ОписаниеОшибки());
	КонецПопытки;

	Возврат ОтветСJSON(Ответ, Результат);

#Область ЗагрузкаЗаказа

// Порт tools/as_order_loader/as_order_loader.bsl (ЗагрузитьЗаказ) + создание
// Характеристики из tools/as_order_loader/build_execute_payload.py.
// Параметры:
//   Результат       — Структура(Успех, НомерЗаказа, Ошибки, Предупреждения), дополняется;
//   Позиции         — Массив соответствий по контракту JSON (article, qty,
//                     thickness, material, shina, conn0, conn1, params, comment);
//   КомментарийЗаказа — Комментарий документа;
//   ИдЗапроса       — request_id для идемпотентности (ТЗ п. 6), может быть пустым.
//
Процедура ЗагрузитьЗаказ(Результат, Позиции, КомментарийЗаказа, ИдЗапроса)

	// --- Идемпотентность: ранее созданный документ по тому же request_id ---
	Если ЗначениеЗаполнено(ИдЗапроса) Тогда
		РаннийДокумент = НайтиДокументПоИдЗапроса(ИдЗапроса);
		Если РаннийДокумент <> Неопределено Тогда
			Результат.Успех = Истина;
			Результат.НомерЗаказа = РаннийДокумент.Номер;
			Результат.Предупреждения.Добавить(
				"Найден ранее созданный документ по request_id """ + ИдЗапроса + """");
			Возврат;
		КонецЕсли;
	КонецЕсли;

	НачатьТранзакцию();
	Попытка
		ДокОбъект = Документы.асСпецификацияЗаказа.СоздатьДокумент();
		ДокОбъект.Дата = ТекущаяДатаСеанса();
		ДокОбъект.Комментарий = КомментарийЗаказа
			+ ?(ЗначениеЗаполнено(ИдЗапроса), " [req:" + СокрЛП(ИдЗапроса) + "]", "");

		НомерСтроки = 0;
		Для Каждого П Из Позиции Цикл
			НомерСтроки = НомерСтроки + 1;
			Если ТипЗнч(П) <> Тип("Соответствие") Тогда
				Результат.Ошибки.Добавить("Строка " + НомерСтроки + ": элемент спецификации не является объектом");
				Продолжить;
			КонецЕсли;
			ЗаполнитьСтроку(ДокОбъект, П, НомерСтроки, Результат);
		КонецЦикла;

		// Площади, массы и цены — одним проходом штатной обработки (паритет с кнопкой «Расчёт»).
		РассчитатьВсеСтроки(ДокОбъект, Результат);

		ДокОбъект.Записать();
		ЗафиксироватьТранзакцию();
	Исключение
		ОтменитьТранзакцию();
		ВызватьИсключение ОписаниеОшибки();
	КонецПопытки;

	Результат.Успех = (Результат.Ошибки.Количество() = 0);
	Результат.НомерЗаказа = ДокОбъект.Номер;

Процедура ЗаполнитьСтроку(ДокОбъект, П, НомерСтроки, Результат)

	Ошибки = Результат.Ошибки;
	Стр = ДокОбъект.Товары.Добавить();

	// Продукт — ссылка на Справочник.асПродукция (обязательно! не Номенклатура).
	Артикул = Свойство(П, "article");
	Продукт = Справочники.асПродукция.НайтиПоРеквизиту("Артикул", Артикул);
	Если Не ЗначениеЗаполнено(Продукт) Тогда
		Ошибки.Добавить("Строка " + НомерСтроки + ": не найден продукт с артикулом """ + Артикул + """");
	Иначе
		Стр.Продукт = Продукт;
		Стр.Номенклатура = Продукт.Продукт;
		Стр.АртикулПродукции = Артикул;
		ПозДефиса = СтрНайти(Артикул, "-");
		Стр.АртикулГруппы = ?(ПозДефиса > 0, Лев(Артикул, ПозДефиса - 1), Артикул);
		Стр.ГруппаПродукцииСтр = Продукт.ГруппаПродукции;
		Стр.СечениеСтр = Продукт.ВидСечения;
	КонецЕсли;

	Количество = ЧислоСвойства(П, "qty", 1);
	Стр.n = Количество;
	Стр.np = Количество;

	Толщина = ЧислоСвойства(П, "thickness", 0);
	Стр.t = Толщина;
	Стр.t2 = Толщина;

	// Материал — точное наименование из справочника асМатериалы (два знака толщины!).
	ИмяМатериала = Свойство(П, "material");
	Стр.Материал = Справочники.асМатериалы.НайтиПоНаименованию(ИмяМатериала, Истина);
	Стр.Материал2 = Стр.Материал;
	Если Не ЗначениеЗаполнено(Стр.Материал) Тогда
		Ошибки.Добавить("Строка " + НомерСтроки + ": не найден материал """ + ИмяМатериала + """");
	КонецЕсли;

	Стр.КомментарийКПродукцииИзOnLine = Свойство(П, "comment");

	// Шина на соединение: 65/95/105 по правилам техотдела (≤350 / 400–950 / ≥1000).
	// Для нержавеющих изделий с шиной 65 — нерж-вариант составного соединения.
	Шина = Свойство(П, "shina");
	Если ЗначениеЗаполнено(Шина) Тогда
		Нерж = СтрНайти(ИмяМатериала, "нерж") > 0;
		Если Нерж И Шина = "65" Тогда
			ИмяШины = "Шина 20(нерж.430) / УГФ 65(нерж.430)";
		Иначе
			ИмяШины = ?(Шина = "105", "Шина 30 / УГФ 105", "Шина 20 / УГФ " + Шина);
		КонецЕсли;
		ШинаСсылка = Справочники.асПродукция.НайтиПоНаименованию(ИмяШины, Истина);
		Стр.МатериалНаСоединение0 = ШинаСсылка;
		Стр.МатериалНаСоединение1 = ШинаСсылка;
		Если Не ЗначениеЗаполнено(ШинаСсылка) Тогда
			Ошибки.Добавить("Строка " + НомерСтроки + ": не найдена шина """ + ИмяШины + """");
		КонецЕсли;
	КонецЕсли;

	// Соединения: 6 = "Шина / уголок", 2 = "Без соединения".
	УстановитьСоединение(Стр, Свойство(П, "conn0"), "c0", НомерСтроки, Ошибки);
	УстановитьСоединение(Стр, Свойство(П, "conn1"), "c1", НомерСтроки, Ошибки);

	// Размеры и прочие параметры строки.
	Параметры = П.Получить("params");
	Если ТипЗнч(Параметры) = Тип("Соответствие") Тогда
		Для Каждого КЗ Из Параметры Цикл
			Попытка
				Стр[КЗ.Ключ] = КЗ.Значение;
			Исключение
				Результат.Предупреждения.Добавить("Строка " + НомерСтроки
					+ ": параметр """ + КЗ.Ключ + """ не применён");
			КонецПопытки;
		КонецЦикла;
	КонецЕсли;

	// Характеристика: найти существующую или создать (обход бага поиска
	// в «Оформить заказ» — без заполненной характеристики оформление падает
	// «Рабочее наименование не уникально», см. docs/BACKLOG.md).
	// ВНИМАНИЕ: в as_order_loader.bsl этого блока нет — держать в синхроне
	// с tools/as_order_loader/build_execute_payload.py.
	Если ЗначениеЗаполнено(Стр.Номенклатура) Тогда
		МатХ = Строка(Стр.Материал2.НаименованиеДляПечати);
		D_ = Формат(Стр.D0, "ЧГ=");
		A_ = Формат(Стр.A0, "ЧГ=");
		B_ = Формат(Стр.B0, "ЧГ=");
		A1_ = Формат(Стр.A1, "ЧГ=");
		B1_ = Формат(Стр.B1, "ЧГ=");
		L_ = Формат(Стр.L0, "ЧГ=");
		Канд = Новый Массив;
		Если Стр.D0 > 0 Тогда
			Если Стр.L0 > 0 Тогда
				Канд.Добавить("Ф" + D_ + "-" + L_ + "  " + МатХ);
			КонецЕсли;
			Канд.Добавить("Ф" + D_ + "  " + МатХ);
		КонецЕсли;
		Если Стр.A0 > 0 И Стр.B0 > 0 Тогда
			Если Стр.A1 > 0 И Стр.B1 > 0 Тогда
				Если Стр.L0 > 0 Тогда
					Канд.Добавить(A_ + "x" + B_ + "/" + A1_ + "x" + B1_ + "-" + L_ + "  " + МатХ);
				КонецЕсли;
				Канд.Добавить(A_ + "x" + B_ + "/" + A1_ + "x" + B1_ + "  " + МатХ);
			КонецЕсли;
			Если Стр.L0 > 0 Тогда
				Канд.Добавить(A_ + "x" + B_ + "-" + L_ + "  " + МатХ);
			КонецЕсли;
			Канд.Добавить(A_ + "x" + B_ + "  " + МатХ);
		КонецЕсли;
		Хар = Неопределено;
		Для Каждого Кандидат Из Канд Цикл
			НайдХ = Справочники.ХарактеристикиНоменклатуры.НайтиПоНаименованию(
				Кандидат, Истина, , Стр.Номенклатура);
			Если ЗначениеЗаполнено(НайдХ) Тогда
				Хар = НайдХ;
				Прервать;
			КонецЕсли;
		КонецЦикла;
		Если Хар = Неопределено И Канд.Количество() > 0 Тогда
			СпрХ = Справочники.ХарактеристикиНоменклатуры.СоздатьЭлемент();
			СпрХ.Владелец = Стр.Номенклатура;
			СпрХ.Наименование = Канд[0];
			СпрХ.НаименованиеПолное = Канд[0];
			СпрХ.Записать();
			Хар = СпрХ.Ссылка;
		КонецЕсли;
		Если Хар <> Неопределено Тогда
			Стр.Характеристика = Хар;
		КонецЕсли;
	КонецЕсли;

	// Производные размеры (a1, b1, d1, p, ...) из регистра настроек продукции.
	// Без этого шага у фасонки (шумоглушители и др.) расчёт площади уходит в минус.
	ПрименитьНастройкиПродукции(Стр, Продукт);

Процедура УстановитьСоединение(Стр, КодСоединения, Суффикс, НомерСтроки, Ошибки)

	Если Не ЗначениеЗаполнено(КодСоединения) Тогда
		Возврат;
	КонецЕсли;

	Если КодСоединения = "6" Тогда
		ИмяСоединения = "Шина / уголок";
	ИначеЕсли КодСоединения = "2" Тогда
		ИмяСоединения = "Без соединения";
	Иначе
		Ошибки.Добавить("Строка " + НомерСтроки + ": неизвестный код соединения """ + КодСоединения + """");
		Возврат;
	КонецЕсли;

	Соединение = Справочники.асСоединениеВоздуховодов.НайтиПоНаименованию(ИмяСоединения, Истина);
	Если Не ЗначениеЗаполнено(Соединение) Тогда
		Ошибки.Добавить("Строка " + НомерСтроки + ": не найдено соединение """ + ИмяСоединения + """");
		Возврат;
	КонецЕсли;

	Стр["СоединениеВоздуховода_" + Суффикс] = Соединение;
